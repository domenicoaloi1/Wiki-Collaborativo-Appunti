class Client {
  + operation() : void

class TableDataGateway {
  + find(id : Integer) : RecordSet
  + findAll() : RecordSet
  + insert(data) : void
  + update(id : Integer, data) : void
  + delete(id : Integer) : void
}

class RecordSet {
  + data
}

' --- RELAZIONI ---

' Il Client utilizza il Gateway per interagire con la persistenza
Client ..> TableDataGateway : <<use>>

' Il Gateway produce o popola il RecordSet
TableDataGateway ..> RecordSet : <<create>>

note bottom of TableDataGateway
